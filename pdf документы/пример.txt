Рисунок 2 — ERD (схема базы данных)
plantuml@startuml
skinparam linetype ortho
skinparam backgroundColor white
skinparam EntityBackgroundColor #FEFEFE
skinparam EntityBorderColor #333333

entity "users" as users {
  * id : SERIAL <<PK>>
  --
  * username : VARCHAR(100)
  * email : VARCHAR(100)
  full_name : VARCHAR(200)
  * hashed_password : TEXT
  * role : VARCHAR(20)
  * is_active : BOOLEAN
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

entity "property_types" as ptypes {
  * id : SERIAL <<PK>>
  --
  * name : VARCHAR(100)
  description : TEXT
  created_at : TIMESTAMP
}

entity "properties" as props {
  * id : SERIAL <<PK>>
  --
  * type_id : INTEGER <<FK>>
  * area : NUMERIC(10,2)
  finishing : VARCHAR(100)
  wall_material : VARCHAR(100)
  floor : INTEGER
  * price : NUMERIC(15,2)
  discount_price : NUMERIC(15,2)
  * status : VARCHAR(50)
  address : VARCHAR(500)
  description : TEXT
  * is_active : BOOLEAN
  created_at : TIMESTAMP
  updated_at : TIMESTAMP
}

entity "photos" as photos {
  * id : SERIAL <<PK>>
  --
  * property_id : INTEGER <<FK>>
  * file_path : VARCHAR(500)
  description : VARCHAR(300)
  created_at : TIMESTAMP
}

entity "construction_docs" as docs {
  * id : SERIAL <<PK>>
  --
  * property_id : INTEGER <<FK>>
  * name : VARCHAR(300)
  * doc_type : VARCHAR(50)
  * file_path : VARCHAR(500)
  created_at : TIMESTAMP
}

entity "contracts" as contracts {
  * id : SERIAL <<PK>>
  --
  * contract_number : VARCHAR
  * property_id : INTEGER <<FK>>
  created_by : INTEGER <<FK>>
  * buyer_name : VARCHAR(300)
  buyer_passport : VARCHAR(100)
  seller_info : TEXT
  * price : NUMERIC(15,2)
  * contract_date : DATE
  * status : VARCHAR(50)
  notes : TEXT
  created_at : TIMESTAMP
}

entity "audit_logs" as audit {
  * id : SERIAL <<PK>>
  --
  user_id : INTEGER <<FK>>
  * action : VARCHAR(50)
  * table_name : VARCHAR(50)
  record_id : INTEGER
  old_values : TEXT
  new_values : TEXT
  ip_address : VARCHAR(45)
  user_agent : TEXT
  created_at : TIMESTAMP
}

ptypes ||--o{ props : "type_id"
props ||--o{ photos : "property_id"
props ||--o{ docs : "property_id"
props ||--o{ contracts : "property_id"
users ||--o{ contracts : "created_by"
users ||--o{ audit : "user_id"

@enduml

Рисунок Б.1 — Алгоритм авторизации и выдачи JWT
plantuml@startuml
skinparam backgroundColor white
skinparam ArrowColor #333333
skinparam ActivityBorderColor #333333
skinparam ActivityBackgroundColor #FEFEFE
skinparam ActivityDiamondBackgroundColor #FFF8DC
skinparam ActivityStartColor #333333
skinparam ActivityEndColor #333333

start

:Пользователь открывает страницу входа /;
:Вводит email и пароль;
:Django отправляет POST /api/v1/auth/login\n(email, password);

:FastAPI получает запрос;
:Поиск пользователя в таблице users по email;

if (Пользователь найден?) then (нет)
  :Вернуть 401 Unauthorized\n«Неверный email или пароль»;
  :Показать сообщение об ошибке;
  stop
else (да)
  :Проверить пароль через bcrypt;
  if (Пароль совпадает?) then (нет)
    :Вернуть 401 Unauthorized\n«Неверный email или пароль»;
    :Показать сообщение об ошибке;
    stop
  else (да)
    :Сгенерировать access_token\n(срок жизни 30 минут, sub = user_id, role);
    :Сгенерировать refresh_token\n(срок жизни 24 часа);
    :Вернуть токены клиенту в формате JSON;
    :Django сохраняет в сессии:\naccess_token, email, role,\nfull_name, user_id;

    if (Роль пользователя?) then (admin)
      :Редирект → admin_dashboard;
    elseif (manager) then (manager)
      :Редирект → manager_dashboard;
    elseif (user) then (user)
      :Редирект → user_home;
    else (external)
      :Редирект → public_catalog;
    endif

    :В последующих запросах Django подставляет\nAuthorization: Bearer <access_token>;
    :FastAPI декодирует токен,\nизвлекает user_id и role;

    if (Токен действителен?) then (нет)
      :Вернуть 401, сбросить сессию Django;
      :Редирект на страницу входа;
      stop
    else (да)
      :Выполнить запрошенную операцию;
    endif
  endif
endif

stop
@enduml

Рисунок Б.2 — Алгоритм добавления объекта недвижимости с фотографиями
plantuml@startuml
skinparam backgroundColor white
skinparam ArrowColor #333333
skinparam ActivityBorderColor #333333
skinparam ActivityBackgroundColor #FEFEFE
skinparam ActivityDiamondBackgroundColor #FFF8DC
skinparam ActivityStartColor #333333
skinparam ActivityEndColor #333333

start

:Администратор авторизуется\n(роль admin, проверка через _resolve_admin);
:Переход на /desk/admin/properties/new/;
:Загрузка списка типов объектов\nчерез property_type_controller;
:Отображение формы добавления объекта;

:Пользователь заполняет форму:\n— тип объекта\n— площадь, этаж\n— материал стен, тип отделки\n— основная цена\n— скидочная цена (опционально)\n— адрес;

:Отправка POST-запроса;
:Вызов property_controller.parse_property_create_post\n(парсинг и валидация полей);

if (Данные валидны?) then (нет)
  :Форма отображается повторно\nс сообщениями об ошибках;
  :Пользователь исправляет данные;
  stop
else (да)
  :FastAPI создаёт запись в таблице properties\nсо статусом for_sale;
  :Запись в audit_logs:\nдействие CREATE_PROPERTY,\nID объекта, временная метка;
  :API возвращает 201 Created\nи данные созданного объекта;
  :Редирект на страницу загрузки фотографий\nadmin_property_photos_view;

  repeat
    :Администратор выбирает файл фотографии;
    :Отправка POST /api/v1/properties/{id}/photos/;
    :Валидация файла через Pillow\n(формат, размер, тип);

    if (Файл валиден?) then (нет)
      :Показать сообщение об ошибке\nчерез messages.error;
    else (да)
      :Сохранение файла на сервере;
      :Создание записи в таблице photos;
      :Запись в audit_logs:\nдействие ADD_PHOTO;
      :Показать сообщение об успехе;
    endif

  repeat while (Загрузить ещё фото?) is (да)
  -> нет;

  :Объект недвижимости с фотографиями создан;
  :Редирект на карточку объекта;
endif

stop
@enduml

Рисунок Б.3 — Алгоритм формирования договора купли-продажи
plantuml@startuml
skinparam backgroundColor white
skinparam ArrowColor #333333
skinparam ActivityBorderColor #333333
skinparam ActivityBackgroundColor #FEFEFE
skinparam ActivityDiamondBackgroundColor #FFF8DC
skinparam ActivityStartColor #333333
skinparam ActivityEndColor #333333

start

:Пользователь с ролью admin или manager\nавторизуется в системе;
:Переход на /desk/admin/contracts/;

if (Договор уже создан?) then (нет)
  :Нажать «Создать договор»;
  :Заполнить форму:\n— выбрать объект недвижимости\n— ввести ФИО покупателя\n— ввести паспортные данные покупателя\n— указать цену договора\n— указать дату заключения;
  :POST /api/v1/contracts/;

  if (Данные валидны?) then (нет)
    :Форма отображается повторно\nс сообщениями об ошибках;
    stop
  else (да)
    :FastAPI создаёт запись в таблице contracts;
    :Запись в audit_logs:\nдействие CREATE_CONTRACT;
  endif
else (да)
  :Выбрать существующий договор;
endif

:Пользователь нажимает «Скачать договор»;
:Выбор формата документа;

if (Выбранный формат?) then (Word)
  :GET /desk/admin/contracts/{id}/word/;
else (Excel)
  :GET /desk/admin/contracts/{id}/excel/;
endif

:Вызов contract_export_controller.export_contract;
:GET-запрос в FastAPI для получения данных\nдоговора и объекта недвижимости;
:FastAPI выполняет запрос к БД\n(таблицы contracts + properties);

if (Данные получены?) then (нет)
  :Вернуть ошибку;
  :Показать сообщение через messages.error;
  :Редирект на /desk/admin/contracts/;
  stop
else (да)
  :Генерация документа:\n— данные сторон (продавец, покупатель)\n— характеристики объекта\n— цена и условия оплаты\n— условия передачи имущества\n— подписи сторон;

  if (Формат?) then (Word)
    :Формирование .docx через python-docx;
    :Имя файла: contract_{date}_{number}.docx;
  else (Excel)
    :Формирование .xlsx через openpyxl;
    :Имя файла: contract_{date}_{number}.xlsx;
  endif

  :Django оборачивает файл в FileResponse;
  :Браузер загружает готовый файл;
  :Запись в audit_logs:\nдействие EXPORT_CONTRACT;
endif

stop
@enduml
